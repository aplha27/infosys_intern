    <script src="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/js/bootstrap.bundle.min.js"></script>
    <script>
        const startBtn = document.getElementById('startBtn');
        const nextBtn = document.getElementById('nextBtn');
        const currentQuestion = document.getElementById('currentQuestion');
        const transcribedAnswer = document.getElementById('transcribedAnswer');
        const statusIndicator = document.getElementById('statusIndicator');
        const progressBar = document.getElementById('progressBar');
        const recordingStatus = document.getElementById('recordingStatus');
        const questionNumber = document.getElementById('questionNumber');

        let interviewStarted = false;
        let currentQuestionIndex = 0;

        startBtn.addEventListener('click', async () => {
            try {
                startBtn.disabled = true;
                statusIndicator.className = 'status-indicator status-active';
                
                const response = await fetch('/start_interview', {
                    method: 'POST'
                });
                
                const data = await response.json();
                if (data.error) {
                    throw new Error(data.error);
                }
                
                currentQuestion.textContent = data.question;
                nextBtn.disabled = false;
                interviewStarted = true;
                currentQuestionIndex = 1;
                questionNumber.textContent = currentQuestionIndex;
                
            } catch (error) {
                console.error('Error:', error);
                currentQuestion.textContent = 'Error starting interview. Please refresh and try again.';
                startBtn.disabled = false;
                statusIndicator.className = 'status-indicator status-inactive';
            }
        });

        nextBtn.addEventListener('click', async () => {
            try {
                nextBtn.disabled = true;
                recordingStatus.style.display = 'block';
                transcribedAnswer.textContent = 'Recording your answer...';
                
                const response = await fetch('/next_question', {
                    method: 'POST'
                });
                
                const data = await response.json();
                recordingStatus.style.display = 'none';
                
                if (data.error) {
                    throw new Error(data.error);
                }
                
                if (data.complete) {
                    // Interview is complete
                    currentQuestion.textContent = data.message;
                    transcribedAnswer.textContent = 'Interview completed!';
                    nextBtn.disabled = true;
                    statusIndicator.className = 'status-indicator status-inactive';
                    progressBar.style.width = '100%';
                    questionNumber.textContent = '5';
                } else {
                    // Show next question
                    currentQuestion.textContent = data.question;
                    transcribedAnswer.textContent = data.last_answer;
                    nextBtn.disabled = false;
                    progressBar.style.width = `${data.progress}%`;
                    currentQuestionIndex++;
                    questionNumber.textContent = currentQuestionIndex;
                }
                
            } catch (error) {
                console.error('Error:', error);
                recordingStatus.style.display = 'none';
                transcribedAnswer.textContent = 'Error processing answer. Please try again.';
                nextBtn.disabled = false;
            }
        });
    </script>
